\begin{definition}[Gini coefficient]
For incomes $y_1,\dots,y_n$ with mean $\mu$,
\begin{equation}
G = \frac{1}{2n^2\mu}\sum_{i=1}^{n}\sum_{j=1}^{n}\lvert y_i - y_j\rvert .
\label{eq:gini}
\end{equation}
\end{definition}
